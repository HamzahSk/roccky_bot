```js
}

const normalizeButton = (btn) =>
   typeof btn === 'string' ? { type: 'reply', displayText: btn, id: btn } : btn

const buttonToNativeFlow = (btn) => {
   btn = normalizeButton(btn)
   if (!btn || typeof btn !== 'object') return null
   const type = (btn.type || btn.name || '').toLowerCase()
   const label = btn.displayText || btn.text || btn.title || ''
   if (type === 'reply' || type === 'quick_reply')
      return { name: 'quick_reply', buttonParamsJson: JSON.stringify({ display_text: label, id: btn.id ?? btn.value ?? '' }) }
   if (type === 'url' || type === 'cta_url')
      return { name: 'cta_url', buttonParamsJson: JSON.stringify({ display_text: label, url: btn.url ?? '', webview_interaction: btn.webview || false }) }
   if (type === 'call' || type === 'cta_call')
      return { name: 'cta_call', buttonParamsJson: JSON.stringify({ display_text: label, id: btn.phoneNumber ?? btn.id ?? '' }) }
   if (type === 'list' || type === 'single_select' || type === 'sections')
      return { name: 'single_select', buttonParamsJson: JSON.stringify({
         title: btn.title || label,
         sections: (btn.sections || []).map(s => ({
            title: s.title || '',
            rows: (s.rows || []).map(r => ({ title: r.title || '', description: r.description || '', id: r.id ?? r.title ?? '' }))
         }))
      }) }
   if (btn.name && btn.buttonParamsJson)
      return { name: btn.name, buttonParamsJson: typeof btn.buttonParamsJson === 'string' ? btn.buttonParamsJson : JSON.stringify(btn.buttonParamsJson) }
   return null
}

const normalizeQuoted = (quoted) => {
   if (!quoted) return undefined
   if (quoted.key && quoted.message) return quoted
   if (quoted.key && quoted.msg) return { key: quoted.key, message: { [quoted.type || 'extendedTextMessage']: quoted.msg } }
   return quoted
}

const toFallbackText = ({ text = '', footer = '', buttons = [] }) => {
   const lines = buttons.map((b, i) => {
      b = normalizeButton(b) || {}
      const label = b.displayText || b.text || b.title || ''
      const value = b.id ?? b.url ?? b.phoneNumber ?? ''
      return `  ${i + 1}. ${label}${value ? ` — ${value}` : ''}`
   }).filter(Boolean).join('\n')
   return [text, footer, lines].filter(Boolean).join('\n\n')
}

const relayInteractive = async (sock, jid, interactiveMessage, { quoted = null, fallback = true, fallbackText = '', relayOptions = {} } = {}) => {
   try {
      const message = generateWAMessageFromContent(jid, { interactiveMessage }, { quoted: normalizeQuoted(quoted), userJid: sock?.user?.id })
      return await sock.relayMessage(jid, message.message, {
         messageId: message.key.id,
         additionalNodes: buildInteractiveNodes(jid),
         ...relayOptions
      })
   } catch (error) {
      console.error('❌ relayInteractive error:', error)
      if (!fallback) throw error
      return sock.sendText(jid, fallbackText, quoted)
   }
}

const buildNativeFlow = (buttons, messageParams = {}) => {
   const nativeFlowButtons = (buttons || []).map(buttonToNativeFlow).filter(Boolean)
   if (!nativeFlowButtons.length) return null
   return {
      buttons: nativeFlowButtons,
      ...(messageParams && Object.keys(messageParams).length ? { messageParamsJson: JSON.stringify(messageParams) } : {}),
      messageVersion: 1
   }
}

const prepareMedia = async (sock, kind, media) => {
   if (!media) return null
   try {
      const source = typeof media === 'string' ? { [kind]: { url: media } } : { [kind]: media }
      return await prepareWAMessageMedia(source, { upload: sock.waUploadToServer })
   } catch (e) { console.error('❌ prepareMedia error:', e); return null }
}

// ===== PUBLIC API =====
Func.sendInteractiveMessage = async (sock, jid, options = {}) => {
   const { text = '', footer = '', header = '', buttons = [], contextInfo = {}, messageParams = {}, quoted = null, fallback = true, relayOptions = {} } = options
   const nativeFlowMessage = buildNativeFlow(buttons, messageParams)
   if (!nativeFlowMessage) return sock.sendText(jid, text, quoted, { footer })
   const interactiveMessage = {
      body: { text: String(text) },
      ...(footer ? { footer: { text: String(footer) } } : {}),
      header: { title: String(header || ''), hasMediaAttachment: false },
      contextInfo: contextInfo || {},
      nativeFlowMessage
   }
   return relayInteractive(sock, jid, interactiveMessage, { quoted, fallback, fallbackText: toFallbackText({ text, footer, buttons }), relayOptions })
}

Func.sendButton = async (sock, jid, options = {}) => {
   const { text, footer, header, buttons = [], ...rest } = options
   return Func.sendInteractiveMessage(sock, jid, { text, footer, header, buttons, ...rest })
}

Func.sendList = async (sock, jid, options = {}) => {
   const { title = 'Pilih Opsi', buttonText = 'Pilih', footer = '', text = '', sections = [], ...rest } = options
   return Func.sendInteractiveMessage(sock, jid, {
      text: text || title, footer,
      buttons: [{ type: 'list', title, displayText: buttonText, sections }],
      ...rest
   })
}
Func.sendSections = async (sock, jid, options = {}) => Func.sendList(sock, jid, options)

Func.sendCard = async (sock, jid, options = {}) => {
   const { text = '', footer = '', header = '', image, video, buttons = [], contextInfo = {}, messageParams = {}, quoted = null, fallback = true, relayOptions = {} } = options
   const interactiveMessage = {
      body: { text: String(text) },
      ...(footer ? { footer: { text: String(footer) } } : {}),
      contextInfo: contextInfo || {},
      header: { title: String(header || ''), hasMediaAttachment: false }
   }
   const media = image ? await prepareMedia(sock, 'image', image) : video ? await prepareMedia(sock, 'video', video) : null
   if (media) interactiveMessage.header = { title: String(header || ''), hasMediaAttachment: true, ...media }
   const nativeFlowMessage = buildNativeFlow(buttons, messageParams)
   if (nativeFlowMessage) interactiveMessage.nativeFlowMessage = nativeFlowMessage
   return relayInteractive(sock, jid, interactiveMessage, { quoted, fallback, fallbackText: toFallbackText({ text, footer, buttons }), relayOptions })
}

Func.sendCarousel = async (sock, jid, options = {}) => {
   const { text = '', footer = '', cards = [], contextInfo = {}, quoted = null, fallback = true, relayOptions = {} } = options
   if (!cards.length) throw new Error('sendCarousel membutuhkan minimal 1 kartu')
   const processedCards = []
   for (const card of cards) {
      const media = card.image ? await prepareMedia(sock, 'image', card.image) : card.video ? await prepareMedia(sock, 'video', card.video) : null
      const cardMessage = {
         body: { text: String(card.text || '') },
         ...(card.footer ? { footer: { text: String(card.footer) } } : {}),
         header: { title: String(card.title || ''), hasMediaAttachment: !!media, ...(media || {}) }
      }
      const nf = buildNativeFlow(card.buttons || [], card.messageParams || {})
      if (nf) cardMessage.nativeFlowMessage = nf
      processedCards.push(cardMessage)
   }
   const interactiveMessage = {
      body: { text: String(text) }, ...(footer ? { footer: { text: String(footer) } } : {}),
      header: { title: '', hasMediaAttachment: false }, contextInfo: contextInfo || {},
      carouselMessage: { cards: processedCards, messageVersion: 1, carouselCardType: 1 }
   }
   const fallbackText = [text, footer, ...cards.map((c, i) => `  ${i + 1}. ${c.title || ''}${c.text ? ` — ${c.text}` : ''}`)].filter(Boolean).join('\n\n')
   return relayInteractive(sock, jid, interactiveMessage, { quoted, fallback, fallbackText, relayOptions })
}
```

**Integrasi:** `lib/SocketClient.js` bind ke `sock.sendButton/sendList/sendSections/sendCard/sendCarousel/sendInteractiveMessage`; `lib/Listener.js` menambahkan semua nama itu langsung ke context plugin (`run(m, { sendButton, sendList, ... })`).
